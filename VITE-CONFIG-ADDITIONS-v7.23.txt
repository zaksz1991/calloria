  calloriaDigitalExperienceSuitePluginV722()

Keep that registration. Add this import alongside the other build-plugin imports:

import { calloriaCustomerRecoveryPlanPluginV723 } from './build/calloria-customer-recovery-plan-v723.mjs';

calloriaDigitalExperienceSuitePluginV722(),
